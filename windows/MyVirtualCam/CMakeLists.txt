cmake_minimum_required(VERSION 3.24)
project(MyVirtualCam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT BASECLASSES)
  message(FATAL_ERROR "Pass -DBASECLASSES=<Microsoft DirectShow baseclasses directory>")
endif()
file(GLOB BASE_SOURCES "${BASECLASSES}/*.cpp")
add_library(strmbase STATIC ${BASE_SOURCES})
target_include_directories(strmbase PUBLIC "${BASECLASSES}")
target_compile_definitions(strmbase PUBLIC UNICODE _UNICODE WIN32 NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(strmbase PRIVATE /wd4311 /wd4312 /wd4996 /wd4244 /wd4267)
target_link_libraries(strmbase PUBLIC strmiids winmm ole32 oleaut32 uuid advapi32)
add_library(MyVirtualCam SHARED MyVirtualCam.cpp MyVirtualCam.def)
target_link_libraries(MyVirtualCam PRIVATE strmbase)
add_executable(MyVirtualCamTest MyVirtualCamTest.cpp)
target_link_libraries(MyVirtualCamTest PRIVATE strmbase)
